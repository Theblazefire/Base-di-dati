-- --------------------------------------------------------
-- Struttura della tabella esibizioni
-- --------------------------------------------------------
CREATE TABLE esibizioni (
  id_artista BIGINT NOT NULL,
  id_brano BIGINT NOT NULL,
  data DATE NOT NULL,
  PRIMARY KEY (id_artista, id_brano, data),
  CONSTRAINT esibizioni_ibfk_1 FOREIGN KEY (id_artista) REFERENCES artisti (id_artista) ON DELETE CASCADE,
  CONSTRAINT esibizioni_ibfk_2 FOREIGN KEY (id_brano) REFERENCES brani (id_brano) ON DELETE CASCADE
);
